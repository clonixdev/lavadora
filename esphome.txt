# --- Pinout UART ESP-01 (atalaya06) <-> Arduino UNO (9600 8N1, GND comun) ---
# (Rama git "wifi": el sketch leia JSON solo por espSerial; no habia esphome en ese arbol. Mismo NeoSWSerial A1/A2.)
# El log ">>> ">PING\n"" confirma que el ESP ENVIA por el pin configurado como tx_pin (GPIO1).
# Si "UART comandos '>' recibidos" no sube y no ves "<<< OK PONG", el Arduino no recibe en A1.
#
# MUY IMPORTANTE (etiquetas del modulo ESP8266 / ESP-01):
#   El dato hacia el Arduino sale por el pad **TX del modulo**, que en UART0 es **GPIO1**
#   (no uses el pad **RX / GPIO3** para llevar comandos al Arduino: ese es solo ESP<-Arduino).
#   Es decir: cable continuo del **TX del ESP (GPIO1)** al **A1 (RX del NeoSWSerial)**.
#   Arduino A2 (TX NeoSWSerial) -> pad **RX del ESP (GPIO3)**.
# Si cruzaste TX-TX (GPIO1-A2) y RX-RX (GPIO3-A1), el telemetria puede ir bien y los comandos no.
#
# Nivel logico: UNO a 5 V; el ESP manda ~3,3 V. Si tras verificar pads el Rx sigue en 0, medir con
# multimetro/osciloscopio tension en A1 al pulsar PING (debe cambiar). A veces hace falta
# nivelador 3,3<->5 V o pull-up debil a 5 V en RX del UNO (consultar esquema seguro antes).
#
# UART0 del ESP8266 puede alternar a tx_pin: GPIO15 / rx_pin: GPIO13 (ver doc ESPHome UART);
# en ESP-01 a menudo solo salen GPIO1/GPIO3 — el swap suele requerir otra placa con esos pines.

uart:
  id: uart_bus
  tx_pin: GPIO1
  rx_pin: GPIO3
  baud_rate: 9600
  debug:
    direction: BOTH
    dummy_receiver: true
    after:
      delimiter: "\r\n"
    sequence:
      - lambda: |-
          if (bytes.empty())
            return;
          {
            bool only_ws = true;
            for (uint8_t b : bytes) {
              if (b != '\r' && b != '\n' && b != ' ' && b != '\t') {
                only_ws = false;
                break;
              }
            }
            if (only_ws)
              return;
          }
          if (direction == uart::UART_DIRECTION_TX) {
            UARTDebug::log_string(direction, bytes);
            return;
          }
          std::string data(bytes.begin(), bytes.end());
          while (!data.empty() && (unsigned char) data.back() <= 32)
            data.pop_back();
          while (!data.empty() && (unsigned char) data.front() <= 32)
            data.erase(0, 1);
          if (data.size() < 2)
            return;
          if (data[0] == '!' && data[1] == 'E' && data.size() > 2 && data[2] == '|') {
            id(ultimo_error_txt).publish_state(data.substr(3));
            return;
          }
          auto fase_from_code = [](int c) -> const char * {
            switch (c) {
              case 0: return "idle";
              case 1: return "llenado_pre";
              case 2: return "llenado_lavado";
              case 3: return "llenado_suav";
              case 4: return "llenado";
              case 5: return "lavado";
              case 6: return "vaciado";
              case 7: return "centrifugar";
              case 8: return "espera";
              default: return "desconocido";
            }
          };
          if (data[0] == 'R' && data[1] == '|') {
            int rstate, rcode, prog, fa, tf, mn, se, ec, ts;
            if (sscanf(data.c_str(), "R|%d|%d|%d|%d|%d|%d|%d|%d|%d", &rstate, &rcode, &prog, &fa, &tf, &mn, &se, &ec, &ts) == 9) {
              UARTDebug::log_string(direction, bytes);
              id(recovery_pending).publish_state(true);
              id(programa_id).publish_state(prog);
              id(faseactual).publish_state(fa);
              id(total_fases).publish_state(tf);
              id(minuto).publish_state(mn);
              id(segundo).publish_state(se);
              id(last_error_code).publish_state(ec);
              id(tambor_guardado).publish_state(ts);
              const char *rr = "";
              if (rcode == 1)
                rr = "error";
              else if (rcode == 2)
                rr = "power_loss";
              id(recovery_reason).publish_state(rr);
            }
            return;
          }
          if (data[0] != 'S' || data[1] != '|')
            return;
          int enc, fa, tf, tv, mn, se, pa, tt, tr, pid, fcode, rp, rcode, ec, e0, e1, e2, urx;
          if (sscanf(data.c_str(),
                       "S|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d|%d",
                       &enc, &fa, &tf, &tv, &mn, &se, &pa, &tt, &tr, &pid, &fcode, &rp, &rcode, &ec, &e0, &e1, &e2, &urx) != 18)
            return;
          UARTDebug::log_string(direction, bytes);
          id(fase).publish_state(fase_from_code(fcode));
          id(faseactual).publish_state(fa);
          id(minuto).publish_state(mn);
          id(segundo).publish_state(se);
          id(paso_motor).publish_state(pa);
          id(encendida).publish_state(enc != 0);
          id(tambor_vacio).publish_state(tv);
          id(tambor_lleno).publish_state(tv == 0);
          id(tiempo_total).publish_state(tt);
          id(tiempo_restante).publish_state(tr);
          id(total_fases).publish_state(tf);
          id(programa_id).publish_state(pid);
          const char *pn = "Desconocido";
          switch (pid) {
            case 1: pn = "Largo"; break;
            case 2: pn = "Corto"; break;
            case 3: pn = "Vaciado"; break;
            case 4: pn = "Corto 2"; break;
            case 5: pn = "Centrifugar"; break;
            default: break;
          }
          id(programa_nombre).publish_state(pn);
          id(recovery_pending).publish_state(rp != 0);
          const char *rr2 = "";
          if (rcode == 1)
            rr2 = "error";
          else if (rcode == 2)
            rr2 = "power_loss";
          id(recovery_reason).publish_state(rr2);
          id(last_error_code).publish_state(ec);
          id(err_ring_0).publish_state(e0);
          id(err_ring_1).publish_state(e1);
          id(err_ring_2).publish_state(e2);
          id(uart_cmd_rx).publish_state(urx);
          if (tf > 0) {
            float pct = 100.0f * (float) fa / (float) tf;
            if (pct > 100.0f)
              pct = 100.0f;
            id(avance_ciclo_pct).publish_state(pct);
          }
          {
            char buf[160];
            snprintf(buf, sizeof(buf), "%s · Fase %d/%d · %d min fase · %d min rest · %s",
                     fase_from_code(fcode), fa, tf, mn, tr, enc ? "En marcha" : "Parada");
            id(estado_resumen).publish_state(buf);
          }

select:
  - platform: template
    name: "Iniciar programa (lista)"
    id: programa_select
    icon: mdi:format-list-bulleted
    optimistic: true
    options:
      - "---"
      - "Largo"
      - "Corto"
      - "Corto 2"
      - "Vaciado"
      - "Centrifugar"
    initial_option: "---"
    set_action:
      - delay: 15ms
      - if:
          condition:
            lambda: "return id(programa_select).state != std::string(\"---\");"
          then:
            - uart.write:
                id: uart_bus
                data: !lambda |-
                  const std::string &v = id(programa_select).state;
                  char c = 'L';
                  if (v == "Corto") c = 'C';
                  else if (v == "Corto 2") c = '2';
                  else if (v == "Vaciado") c = 'V';
                  else if (v == "Centrifugar") c = 'X';
                  else if (v == "Largo") c = 'L';
                  std::string out = std::string(">START,") + c + "\n";
                  return std::vector<uint8_t>(out.begin(), out.end());
