uart:
  id: uart_bus
  tx_pin: GPIO1
  rx_pin: GPIO3
  baud_rate: 9600
  debug:
    direction: BOTH
    dummy_receiver: true
    after:
      delimiter: "\r\n"
    sequence:
      - lambda: |-
          if (bytes.empty())
            return;
          {
            bool only_ws = true;
            for (uint8_t b : bytes) {
              if (b != '\r' && b != '\n' && b != ' ' && b != '\t') {
                only_ws = false;
                break;
              }
            }
            if (only_ws)
              return;
          }
          if (direction != uart::UART_DIRECTION_RX)
            return;
          std::string data(bytes.begin(), bytes.end());
          while (!data.empty() && (unsigned char) data.back() <= 32)
            data.pop_back();
          while (!data.empty() && (unsigned char) data.front() <= 32)
            data.erase(0, 1);
          if (data.empty() || data.front() != '{')
            return;
          JsonDocument doc;
          DeserializationError err = deserializeJson(doc, data);
          if (err)
            return;
          JsonObject root = doc.as< JsonObject >();
          auto get_int = [&](const char *a, const char *b, int def) -> int {
            if (!root[a].isNull())
              return static_cast<int>(root[a].as< double >());
            if (b && !root[b].isNull())
              return static_cast<int>(root[b].as< double >());
            return def;
          };
          bool looks_like_status = !root["Encendida"].isNull() || !root["FaseActual"].isNull() || !root["faseActual"].isNull() ||
              !root["ProgramaId"].isNull() || !root["TamborVacio"].isNull() || !root["TiempoTotal"].isNull() ||
              !root["recovery_pending"].isNull() || !root["error"].isNull();
          if (!looks_like_status)
            return;
          UARTDebug::log_string(direction, bytes);

          if (!root["Fase"].isNull())
            id(fase).publish_state(root["Fase"].as< const char * >());

          int fa = get_int("FaseActual", "faseActual", 0);
          id(faseactual).publish_state(fa);

          int mn = get_int("Minuto", "minuto", 0);
          id(minuto).publish_state(mn);

          if (!root["Segundo"].isNull() || !root["segundo"].isNull())
            id(segundo).publish_state(get_int("Segundo", "segundo", 0));

          if (!root["Paso"].isNull())
            id(paso_motor).publish_state(static_cast<int>(root["Paso"].as< double >()));

          if (!root["Encendida"].isNull())
            id(encendida).publish_state(root["Encendida"].as< bool >());

          if (!root["TamborVacio"].isNull()) {
            int tv = static_cast<int>(root["TamborVacio"].as< double >());
            id(tambor_vacio).publish_state(tv);
            id(tambor_lleno).publish_state(tv == 0);
          }

          if (!root["TiempoTotal"].isNull())
            id(tiempo_total).publish_state(static_cast<int>(root["TiempoTotal"].as< double >()));

          if (!root["TiempoRestante"].isNull())
            id(tiempo_restante).publish_state(static_cast<int>(root["TiempoRestante"].as< double >()));

          int tf = get_int("totalFases", nullptr, 0);
          id(total_fases).publish_state(tf);

          if (!root["ProgramaId"].isNull() || !root["programa"].isNull()) {
            int pid = !root["ProgramaId"].isNull()
                          ? static_cast<int>(root["ProgramaId"].as< double >())
                          : static_cast<int>(root["programa"].as< double >());
            id(programa_id).publish_state(pid);
            const char *pn = "Desconocido";
            switch (pid) {
              case 1: pn = "Largo"; break;
              case 2: pn = "Corto"; break;
              case 3: pn = "Vaciado"; break;
              case 4: pn = "Corto 2"; break;
              case 5: pn = "Centrifugar"; break;
              default: break;
            }
            id(programa_nombre).publish_state(pn);
          }

          bool rp = false;
          if (!root["recovery_pending"].isNull())
            rp = root["recovery_pending"].as< bool >();
          id(recovery_pending).publish_state(rp);

          const char *rr = nullptr;
          if (!root["recovery_reason"].isNull())
            rr = root["recovery_reason"].as< const char * >();
          else if (!root["reason"].isNull())
            rr = root["reason"].as< const char * >();
          if (rr && rr[0])
            id(recovery_reason).publish_state(rr);

          if (!root["last_error_code"].isNull())
            id(last_error_code).publish_state(static_cast<int>(root["last_error_code"].as< double >()));

          if (!root["ErrRing0"].isNull())
            id(err_ring_0).publish_state(static_cast<int>(root["ErrRing0"].as< double >()));
          if (!root["ErrRing1"].isNull())
            id(err_ring_1).publish_state(static_cast<int>(root["ErrRing1"].as< double >()));
          if (!root["ErrRing2"].isNull())
            id(err_ring_2).publish_state(static_cast<int>(root["ErrRing2"].as< double >()));

          if (!root["error"].isNull())
            id(ultimo_error_txt).publish_state(root["error"].as< const char * >());

          if (!root["tambor_saved"].isNull())
            id(tambor_guardado).publish_state(static_cast<int>(root["tambor_saved"].as< double >()));

          if (tf > 0 && !root["FaseActual"].isNull()) {
            float pct = 100.0f * (float) fa / (float) tf;
            if (pct > 100.0f) pct = 100.0f;
            id(avance_ciclo_pct).publish_state(pct);
          }

          if (!root["Fase"].isNull() && (!root["FaseActual"].isNull() || !root["faseActual"].isNull())) {
            char buf[160];
            int trest = !root["TiempoRestante"].isNull() ? static_cast<int>(root["TiempoRestante"].as< double >()) : 0;
            snprintf(buf, sizeof(buf), "%s · Fase %d/%d · %d min fase · %d min rest · %s",
                     root["Fase"].as< const char * >(), fa, tf, mn, trest,
                     id(encendida).state ? "En marcha" : "Parada");
            id(estado_resumen).publish_state(buf);
          }

    name: "Iniciar programa (lista)"
    id: programa_select
    icon: mdi:format-list-bulleted
    optimistic: true
    options:
      - "---"
      - "Largo"
      - "Corto"
      - "Corto 2"
      - "Vaciado"
      - "Centrifugar"
    initial_option: "---"
    set_action:
      - delay: 15ms
      - if:
          condition:
            lambda: "return id(programa_select).state != std::string(\"---\");"
          then:
            - uart.write: !lambda |-
                const std::string &v = id(programa_select).state;
                std::string p = "largo";
                if (v == "Corto") p = "corto";
                else if (v == "Corto 2") p = "corto2";
                else if (v == "Vaciado") p = "vaciado";
                else if (v == "Centrifugar") p = "centrifugar";
                else if (v == "Largo") p = "largo";
                std::string out = std::string("{\"command\":\"start\",\"programa\":\"") + p + "\"}\n";
                return std::vector<uint8_t>(out.begin(), out.end());
