json:

uart:
  id: uart_bus
  tx_pin: GPIO1
  rx_pin: GPIO3
  baud_rate: 9600
  debug:
    direction: BOTH
    dummy_receiver: true
    after:
      delimiter: "\n"
    sequence:
      - lambda: |-
          UARTDebug::log_string(direction, bytes);
          std::string data(bytes.begin(), bytes.end());
          json::parse_json(data.c_str(), [](JsonObject root) -> bool {
            auto get_int = [&](const char *a, const char *b, int def) -> int {
              if (root.containsKey(a)) return root[a].as<int>();
              if (b && root.containsKey(b)) return root[b].as<int>();
              return def;
            };
            auto get_str = [&](const char *a, const char *b, const char *def) -> const char * {
              if (root.containsKey(a)) return root[a].as<const char *>();
              if (b && root.containsKey(b)) return root[b].as<const char *>();
              return def;
            };

            if (root.containsKey("Fase"))
              id(fase).publish_state(root["Fase"].as<const char *>());

            int fa = get_int("FaseActual", "faseActual", id(faseactual).state);
            id(faseactual).publish_state(fa);

            int mn = get_int("Minuto", "minuto", id(minuto).state);
            id(minuto).publish_state(mn);

            if (root.containsKey("Segundo") || root.containsKey("segundo"))
              id(segundo).publish_state(get_int("Segundo", "segundo", 0));

            if (root.containsKey("Paso"))
              id(paso_motor).publish_state(root["Paso"].as<int>());

            if (root.containsKey("Encendida"))
              id(encendida).publish_state(root["Encendida"].as<bool>());

            if (root.containsKey("TamborVacio")) {
              int tv = root["TamborVacio"].as<int>();
              id(tambor_vacio).publish_state(tv);
              id(tambor_lleno).publish_state(tv == 0);
            }

            if (root.containsKey("TiempoTotal"))
              id(tiempo_total).publish_state(root["TiempoTotal"].as<int>());

            if (root.containsKey("TiempoRestante"))
              id(tiempo_restante).publish_state(root["TiempoRestante"].as<int>());

            int tf = get_int("totalFases", nullptr, id(total_fases).state);
            id(total_fases).publish_state(tf);

            if (root.containsKey("ProgramaId") || root.containsKey("programa")) {
              int pid = root.containsKey("ProgramaId") ? root["ProgramaId"].as<int>() : root["programa"].as<int>();
              id(programa_id).publish_state(pid);
              const char *pn = "Desconocido";
              switch (pid) {
                case 1: pn = "Largo"; break;
                case 2: pn = "Corto"; break;
                case 3: pn = "Vaciado"; break;
                case 4: pn = "Corto 2"; break;
                case 5: pn = "Centrifugar"; break;
                default: break;
              }
              id(programa_nombre).publish_state(pn);
            }

            bool rp = false;
            if (root.containsKey("recovery_pending"))
              rp = root["recovery_pending"].as<bool>();
            id(recovery_pending).publish_state(rp);

            const char *rr = nullptr;
            if (root.containsKey("recovery_reason"))
              rr = root["recovery_reason"].as<const char *>();
            else if (root.containsKey("reason"))
              rr = root["reason"].as<const char *>();
            if (rr && rr[0])
              id(recovery_reason).publish_state(rr);

            if (root.containsKey("last_error_code"))
              id(last_error_code).publish_state(root["last_error_code"].as<int>());

            if (root.containsKey("ErrRing0"))
              id(err_ring_0).publish_state(root["ErrRing0"].as<int>());
            if (root.containsKey("ErrRing1"))
              id(err_ring_1).publish_state(root["ErrRing1"].as<int>());
            if (root.containsKey("ErrRing2"))
              id(err_ring_2).publish_state(root["ErrRing2"].as<int>());

            if (root.containsKey("error"))
              id(ultimo_error_txt).publish_state(root["error"].as<const char *>());

            if (root.containsKey("tambor_saved"))
              id(tambor_guardado).publish_state(root["tambor_saved"].as<int>());

            if (tf > 0 && root.containsKey("FaseActual")) {
              float pct = 100.0f * (float) fa / (float) tf;
              if (pct > 100.0f) pct = 100.0f;
              id(avance_ciclo_pct).publish_state(pct);
            }

            if (root.containsKey("Fase") && (root.containsKey("FaseActual") || root.containsKey("faseActual"))) {
              char buf[160];
              int trest = root.containsKey("TiempoRestante") ? root["TiempoRestante"].as<int>() : id(tiempo_restante).state;
              snprintf(buf, sizeof(buf), "%s · Fase %d/%d · %d min fase · %d min rest · %s",
                       root["Fase"].as<const char *>(), fa, tf, mn, trest,
                       id(encendida).state ? "En marcha" : "Parada");
              id(estado_resumen).publish_state(buf);
            }

            return true;
          });

  - platform: template
    name: "Iniciar programa (lista)"
    id: programa_select
    icon: mdi:format-list-bulleted
    optimistic: true
    options:
      - "---"
      - "Largo"
      - "Corto"
      - "Corto 2"
      - "Vaciado"
      - "Centrifugar"
    initial_option: "---"
    set_action:
      - delay: 15ms
      - if:
          condition:
            lambda: "return id(programa_select).state != std::string(\"---\");"
          then:
            - uart.write: !lambda |-
                const std::string &v = id(programa_select).state;
                std::string p = "largo";
                if (v == "Corto") p = "corto";
                else if (v == "Corto 2") p = "corto2";
                else if (v == "Vaciado") p = "vaciado";
                else if (v == "Centrifugar") p = "centrifugar";
                else if (v == "Largo") p = "largo";
                char buf[96];
                snprintf(buf, sizeof(buf), "{\"command\":\"start\",\"programa\":\"%s\"}\n", p.c_str());
                return std::string(buf);
